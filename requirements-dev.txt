# SweetClaude test dependencies.
# Install with scripts/setup-dev.sh — it also downloads the matching browser
# build, which pip alone does not do.

pytest>=8.0
PyYAML>=6.0

# Coverage measurement and the CI gate (see .coveragerc).
#
# Pinned exactly, for the same reason as playwright below: an unpinned test
# toolchain means CI measures with different code than a developer does.
# Subprocess coverage in particular is carried by pytest-cov's .pth hook, and
# a version difference there silently reports 0% for every script the tests
# drive as a subprocess.
coverage==7.9.1
pytest-cov==4.1.0

# Pinned exactly on purpose. Playwright ships a hard-coded browser revision per
# release and refuses to fall back to an older one, so upgrading the package
# without re-running `playwright install` breaks tests/test_dashboard_ui.py with
# "Executable doesn't exist". Bump this and re-run scripts/setup-dev.sh together.
playwright==1.60.0

# Linter. Pinned exactly for the same reason as coverage: an unpinned ruff
# means CI reports different violations than a developer does. Rule set and
# test-tree exemptions live in ruff.toml; CI runs `ruff check .`.
ruff==0.17.1
